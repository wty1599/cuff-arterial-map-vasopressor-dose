-- Non-executable DuckDB query companion: run its owning guarded Python audit.
-- Requires private registered tables/views; this .sql.txt file is for inspection.
-- Executed with DuckDB over a direct cp1252 read of original @AMSTERDAM_ROOT@/processitems.csv; all process types retained before SQL. No production interval file used.
SELECT a.cid,a.sid,a.t,p.itemid,p.item AS label,p.start/1000.0 AS start_seconds,p.stop/1000.0 AS end_seconds FROM sampled_anchors a LEFT JOIN raw_processitems p ON a.sid=p.admissionid AND lower(p.item) LIKE 'arterielijn %' ORDER BY a.cid,p.start,p.itemid;
