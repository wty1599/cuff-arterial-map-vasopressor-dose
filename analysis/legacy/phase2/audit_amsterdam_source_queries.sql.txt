-- Non-executable DuckDB query companion: run its owning guarded Python audit.
-- Requires private registered tables/views; this .sql.txt file is for inspection.
CREATE OR REPLACE TABLE source_bp AS SELECT x.admissionid sid,CAST(x.measuredat AS DOUBLE)/1000 ts,m.classification modality,m.component,m.family,TRY_CAST(x.value AS DOUBLE) measurement FROM read_csv_auto('@CUFF_WORKSPACE@/phase0/_restricted/amsterdam_numeric_source.csv',all_varchar=true,encoding='latin-1') x JOIN concepts m ON x.itemid=m.itemid JOIN (SELECT sid,min(t)-300 lo,max(t)+300 hi FROM checks GROUP BY sid) z ON x.admissionid=z.sid AND CAST(x.measuredat AS DOUBLE)/1000 BETWEEN z.lo AND z.hi;

SELECT x.*,m.drug FROM read_csv_auto('@AMSTERDAM_ROOT@/drugitems.csv',all_varchar=true,encoding='latin-1') x JOIN vaso_concepts m ON x.itemid=m.itemid WHERE x.admissionid IN (SELECT sid FROM checks) AND x.iscontinuous='1' AND lower(x.ordercategory) LIKE '%spuitpompen%';

SELECT admissionid sid,admittedat,dischargedat,weightgroup,heightgroup FROM read_csv_auto('@AMSTERDAM_ROOT@/admissions.csv',all_varchar=true,encoding='latin-1') WHERE admissionid IN (SELECT sid FROM checks);

CREATE OR REPLACE TABLE source_triplets AS SELECT sid,ts,modality,family,quantile_cont(measurement,.5) FILTER(WHERE component='map') pressure,quantile_cont(measurement,.5) FILTER(WHERE component='sbp') systolic,quantile_cont(measurement,.5) FILTER(WHERE component='dbp') diastolic FROM source_bp GROUP BY sid,ts,modality,family;

CREATE OR REPLACE TABLE source_clean AS SELECT sid,ts,modality,quantile_cont(pressure,.5) pressure FROM source_triplets WHERE pressure>=20 AND pressure<=200 AND (systolic IS NULL OR systolic<=300) AND (systolic IS NULL OR diastolic IS NULL OR (systolic>diastolic AND systolic-diastolic>=10)) GROUP BY sid,ts,modality;

CREATE OR REPLACE TABLE audit_art_window AS SELECT r.audit_id,b.ts,b.pressure,abs(b.ts-r.t) distance FROM reconstructed r JOIN source_clean b ON b.sid=r.sid AND b.modality='arterial' AND b.ts BETWEEN r.t-300 AND r.t+300 AND b.ts>=r.source_start AND b.ts<r.source_stop;

SELECT r.audit_id,r.source_start,r.source_stop,r.source_nee,r.source_clean_nee,r.source_weight,r.source_height,r.source_clean_weight,r.source_clean_height,r.source_clean_bmi,(SELECT quantile_cont(b.pressure,.5) FROM source_clean b WHERE b.sid=r.sid AND b.modality='cuff' AND abs(b.ts-r.t)<.000001) source_cuff,(SELECT quantile_cont(b.pressure,.5) FROM source_clean b WHERE b.sid=r.sid AND b.modality='arterial' AND b.ts BETWEEN r.t-300 AND r.t+300 AND b.ts>=r.source_start AND b.ts<r.source_stop) source_art,(SELECT min(abs(b.ts-r.t)) FROM source_clean b WHERE b.sid=r.sid AND b.modality='arterial' AND b.ts BETWEEN r.t-300 AND r.t+300 AND b.ts>=r.source_start AND b.ts<r.source_stop) source_diff FROM reconstructed r;

SELECT audit_id,count(*) source_art_values,to_json(list(ts) FILTER(WHERE distance=closest)) source_nearest_ties FROM (SELECT *,min(distance) OVER(PARTITION BY audit_id) closest FROM audit_art_window) GROUP BY audit_id;
